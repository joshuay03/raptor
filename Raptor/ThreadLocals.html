def self.clear
  thread = Thread.current
  thread.thread_variables.each do |key|
    thread.thread_variable_set(key, nil) unless PRESERVED_KEYS.include?(key)
  end
end

def self.fetch(key)
  thread = Thread.current
  thread.thread_variable_get(key) || thread.thread_variable_set(key, yield)
end
