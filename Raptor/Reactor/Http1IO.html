def initialize(body, state)
  super()
  @body = body
  @state = state
  @budget = DetachedBody::Budget.new(DETACHED_CONNECTION_BUFFER_SIZE)
  @input = String.new(encoding: Encoding::ASCII_8BIT)
  @reading = true
  @finishing = false
end
